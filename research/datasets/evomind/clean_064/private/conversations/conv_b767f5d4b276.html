```bash
# 先查看现有的 authorized_keys
cat ~/.ssh/authorized_keys

# 然后用 echo 添加完整密钥（替换下面的 YOUR_KEY 为实际内容）

   ```

   ```bash
   ssh-keygen -t ed25519 -C "***@***"

   cat ~/.ssh/authorized_keys
